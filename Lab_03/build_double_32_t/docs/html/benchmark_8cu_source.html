47const char* cublasStatusName(cublasStatus_t status)
48{
49 switch (status) {
50 case CUBLAS_STATUS_SUCCESS: return "CUBLAS_STATUS_SUCCESS";
51 case CUBLAS_STATUS_NOT_INITIALIZED: return "CUBLAS_STATUS_NOT_INITIALIZED";
52 case CUBLAS_STATUS_ALLOC_FAILED: return "CUBLAS_STATUS_ALLOC_FAILED";
53 case CUBLAS_STATUS_INVALID_VALUE: return "CUBLAS_STATUS_INVALID_VALUE";
54 case CUBLAS_STATUS_ARCH_MISMATCH: return "CUBLAS_STATUS_ARCH_MISMATCH";

56 case CUBLAS_STATUS_EXECUTION_FAILED: return "CUBLAS_STATUS_EXECUTION_FAILED";

59 case CUBLAS_STATUS_LICENSE_ERROR: return "CUBLAS_STATUS_LICENSE_ERROR";

93template <typename T>
94void launchCublasGemm(
95 cublasHandle_t handle,
96 int N,
97 int rows,
98 const T* d_A,
99 const T* d_B,
100 T* d_C)
101{
102 const T alpha = T{1};
103 const T beta = T{0};
104
105 if constexpr (std::is_same_v<T, float>) {
106 CUBLAS_CHECK(cublasSgemm(
107 handle,
108 CUBLAS_OP_N,
109 CUBLAS_OP_N,
110 N, rows, N,
111 &alpha,
112 d_B, N,
113 d_A, N,
114 &beta,
115 d_C, N
116 ));
117 }
118 else if constexpr (std::is_same_v<T, double>) {
119 CUBLAS_CHECK(cublasDgemm(
120 handle,
121 CUBLAS_OP_N,
122 CUBLAS_OP_N,
123 N, rows, N,
124 &alpha,
125 d_B, N,
126 d_A, N,
127 &beta,
128 d_C, N
129 ));
130 }
131 else {
132 static_assert(
133 std::is_same_v<T, float> || std::is_same_v<T, double>,
134 "cuBLAS supports only float and double in this benchmark"
135 );
136 }
137}

176cublasHandle_t sharedHandle()
177{
178 struct Holder {
179 cublasHandle_t h = nullptr;
180 Holder() { CUBLAS_CHECK(cublasCreate(&h)); }
181 ~Holder() { if (h != nullptr) { cublasDestroy(h); } }
182 };
183 static Holder holder;
184 return holder.h;
185}

188std::vector<cublasHandle_t>& streamHandlePool(int count)
189{
190 struct Pool {
191 std::vector<cublasHandle_t> h;
192 ~Pool() { for (auto x : h) { if (x != nullptr) { cublasDestroy(x); } } }
193 };
194 static Pool pool;
195 while (static_cast<int>(pool.h.size()) < count) {
196 cublasHandle_t handle = nullptr;
197 CUBLAS_CHECK(cublasCreate(&handle));
198 pool.h.push_back(handle);
199 }
200 return pool.h;
201}

251void launchAlgorithm(
252 Algorithm algorithm,
253 const BASE_TYPE* d_A,
254 const BASE_TYPE* d_B,
255 BASE_TYPE* d_C,
256 int N,
257 int rows,
258 cudaStream_t stream,
259 cublasHandle_t handle = nullptr)
260{
261 const int bs = g_blockSize;
262 const dim3 block(bs, bs);
263 const dim3 grid(
264 (N + bs - 1) / bs,
265 (rows + bs - 1) / bs
266 );
267
268 switch (algorithm) {
269 case Algorithm::Naive:
270 matrixMultNaive<<<grid, block, 0, stream>>>(
271 d_A, d_B, d_C, N, rows
272 );
273 break;
274
275 case Algorithm::StaticTiled:
276 switch (bs) {
277 case 2: launchStaticTiled<2>(grid, stream, d_A, d_B, d_C, N, rows); break;
278 case 4: launchStaticTiled<4>(grid, stream, d_A, d_B, d_C, N, rows); break;
279 case 8: launchStaticTiled<8>(grid, stream, d_A, d_B, d_C, N, rows); break;
280 case 16: launchStaticTiled<16>(grid, stream, d_A, d_B, d_C, N, rows); break;
281 case 32: launchStaticTiled<32>(grid, stream, d_A, d_B, d_C, N, rows); break;
282 default:
283 std::cerr << "Unsupported block size: " << bs << '\n';
284 std::exit(EXIT_FAILURE);
285 }
286 break;
287
288 case Algorithm::DynamicTiled:
289 matrixMultTiledDynamic<<<
290 grid,
291 block,
292 2 * bs * bs * sizeof(BASE_TYPE),
293 stream
294 >>>(d_A, d_B, d_C, N, rows);
295 break;
296
297 case Algorithm::cuBLAS:
298 if (handle == nullptr) {
299 std::cerr << "cuBLAS handle is null\n";
300 std::exit(EXIT_FAILURE);
301 }
302
303 launchCublasGemm<BASE_TYPE>(
304 handle, N, rows, d_A, d_B, d_C
305 );
306 break;
307 }
308
309 CUDA_CHECK(cudaGetLastError());
310}

332Timing runFullMatrix(
333 Algorithm algorithm,
334 MemoryStrategy strategy,
335 const std::vector<BASE_TYPE>& A,
336 const std::vector<BASE_TYPE>& B,
337 std::vector<BASE_TYPE>& C,
338 int N)
339{
340 Timing timing{};
341
342 cublasHandle_t handle = nullptr;
343 if (algorithm == Algorithm::cuBLAS) {
344 handle = sharedHandle();
345 }
346
347 const auto totalStart = Clock::now();
348
349 const std::size_t matrixElements =
350 static_cast<std::size_t>(N) * static_cast<std::size_t>(N);
351 const std::size_t matrixBytes = matrixElements * sizeof(BASE_TYPE);
352
353 const bool usePinned = (strategy == MemoryStrategy::Pinned);
354 const bool useManaged =
355 strategy == MemoryStrategy::UnifiedPrefetch ||
356 strategy == MemoryStrategy::UnifiedManaged;
357
358 BASE_TYPE* d_A = nullptr;
359 BASE_TYPE* d_B = nullptr;
360 BASE_TYPE* d_C = nullptr;
361
362 BASE_TYPE* h_A = const_cast<BASE_TYPE*>(A.data());
363 BASE_TYPE* h_B = const_cast<BASE_TYPE*>(B.data());
364 BASE_TYPE* h_C = C.data();
365
366 BASE_TYPE* pinned_A = nullptr;
367 BASE_TYPE* pinned_B = nullptr;
368 BASE_TYPE* pinned_C = nullptr;
369
370 if (usePinned) {
371 CUDA_CHECK(cudaMallocHost(&pinned_A, matrixBytes));
372 CUDA_CHECK(cudaMallocHost(&pinned_B, matrixBytes));
373 CUDA_CHECK(cudaMallocHost(&pinned_C, matrixBytes));
374
375 std::copy(A.begin(), A.end(), pinned_A);
376 std::copy(B.begin(), B.end(), pinned_B);
377
378 h_A = pinned_A;
379 h_B = pinned_B;
380 h_C = pinned_C;
381 }
382
383 if (useManaged) {
384 CUDA_CHECK(cudaMallocManaged(&d_A, matrixBytes));
385 CUDA_CHECK(cudaMallocManaged(&d_B, matrixBytes));
386 CUDA_CHECK(cudaMallocManaged(&d_C, matrixBytes));
387
388 const auto fillStart = Clock::now();
389 std::copy(A.begin(), A.end(), d_A);
390 std::copy(B.begin(), B.end(), d_B);
391 timing.h2d = elapsedMilliseconds(fillStart, Clock::now());
392
393 if (strategy == MemoryStrategy::UnifiedPrefetch) {
394 const auto prefetchStart = Clock::now();
395
396 CUDA_CHECK(prefetchManaged(d_A, matrixBytes, false));
397 CUDA_CHECK(prefetchManaged(d_B, matrixBytes, false));
398 CUDA_CHECK(prefetchManaged(d_C, matrixBytes, false));
399
400 CUDA_CHECK(cudaDeviceSynchronize());
401
402 timing.h2d += elapsedMilliseconds(prefetchStart, Clock::now());
403 }
404 }
405 else {
406 CUDA_CHECK(cudaMalloc(&d_A, matrixBytes));
407 CUDA_CHECK(cudaMalloc(&d_B, matrixBytes));
408 CUDA_CHECK(cudaMalloc(&d_C, matrixBytes));
409
410 const auto transferStart = Clock::now();
411
412 CUDA_CHECK(cudaMemcpy(
413 d_A, h_A, matrixBytes, cudaMemcpyHostToDevice
414 ));
415 CUDA_CHECK(cudaMemcpy(
416 d_B, h_B, matrixBytes, cudaMemcpyHostToDevice
417 ));
418
419 timing.h2d = elapsedMilliseconds(
420 transferStart, Clock::now()
421 );
422 }
423
424 cudaEvent_t kernelStart{}, kernelStop{};
425 CUDA_CHECK(cudaEventCreate(&kernelStart));
426 CUDA_CHECK(cudaEventCreate(&kernelStop));
427
428 CUDA_CHECK(cudaEventRecord(kernelStart));
429
430 launchAlgorithm(
431 algorithm,
432 d_A, d_B, d_C,
433 N, N,
434 nullptr,
435 handle
436 );
437
438 CUDA_CHECK(cudaEventRecord(kernelStop));
439 CUDA_CHECK(cudaEventSynchronize(kernelStop));
440
441 timing.kernel = eventElapsed(kernelStart, kernelStop);
442
443 const auto downloadStart = Clock::now();
444
445 if (strategy == MemoryStrategy::UnifiedPrefetch) {
446 CUDA_CHECK(prefetchManaged(d_C, matrixBytes, true));
447 CUDA_CHECK(cudaDeviceSynchronize());
448 std::copy(d_C, d_C + matrixElements, C.begin());
449 timing.d2h = elapsedMilliseconds(
450 downloadStart, Clock::now()
451 );
452 }
453 else if (strategy == MemoryStrategy::UnifiedManaged) {
454 CUDA_CHECK(cudaDeviceSynchronize());
455 std::copy(d_C, d_C + matrixElements, C.begin());
456 timing.d2h = elapsedMilliseconds(
457 downloadStart, Clock::now()
458 );
459 }
460 else {
461 CUDA_CHECK(cudaMemcpy(
462 h_C, d_C, matrixBytes, cudaMemcpyDeviceToHost
463 ));
464
465 if (usePinned) {
466 std::copy(pinned_C, pinned_C + matrixElements, C.begin());
467 }
468
469 timing.d2h = elapsedMilliseconds(
470 downloadStart, Clock::now()
471 );
472 }
473
474 CUDA_CHECK(cudaEventDestroy(kernelStart));
475 CUDA_CHECK(cudaEventDestroy(kernelStop));
476
477 CUDA_CHECK(cudaFree(d_A));
478 CUDA_CHECK(cudaFree(d_B));
479 CUDA_CHECK(cudaFree(d_C));
480
481 if (usePinned) {
482 CUDA_CHECK(cudaFreeHost(pinned_A));
483 CUDA_CHECK(cudaFreeHost(pinned_B));
484 CUDA_CHECK(cudaFreeHost(pinned_C));
485 }
486
487 timing.total = elapsedMilliseconds(totalStart, Clock::now());
488 return timing;
489}

517Timing runStreaming(
518 Algorithm algorithm,
519 MemoryStrategy strategy,
520 const std::vector<BASE_TYPE>& A,
521 const std::vector<BASE_TYPE>& B,
522 std::vector<BASE_TYPE>& C,
523 int N)
524{
525 Timing timing{};
526 const auto totalStart = Clock::now();
527
528 const bool usePinned =
529 strategy == MemoryStrategy::PinnedStreams;
530
531 const std::size_t matrixBytes =
532 static_cast<std::size_t>(N) * N * sizeof(BASE_TYPE);
533
534 const int streamCount = std::min(NUM_STREAMS, N);
535 const int rowsPerStream = (N + streamCount - 1) / streamCount;
536
537 BASE_TYPE* d_B = nullptr;
538 BASE_TYPE* pinned_B = nullptr;
539
540 const BASE_TYPE* hostB = B.data();
541
542 std::vector<cudaStream_t> streams(streamCount);
543 std::vector<cublasHandle_t> handles(streamCount, nullptr);
544 if (algorithm == Algorithm::cuBLAS) {
545 auto& pool = streamHandlePool(streamCount);
546 for (int i = 0; i < streamCount; ++i) {
547 handles[i] = pool[i];
548 }
549 }
550
551 std::vector<BASE_TYPE*> d_A(streamCount, nullptr);
552 std::vector<BASE_TYPE*> d_C(streamCount, nullptr);
553
554 std::vector<BASE_TYPE*> pinned_A(streamCount, nullptr);
555 std::vector<BASE_TYPE*> pinned_C(streamCount, nullptr);
556
557 std::vector<cudaEvent_t> h2dStart(streamCount);
558 std::vector<cudaEvent_t> h2dStop(streamCount);
559 std::vector<cudaEvent_t> kernelStart(streamCount);
560 std::vector<cudaEvent_t> kernelStop(streamCount);
561 std::vector<cudaEvent_t> d2hStart(streamCount);
562 std::vector<cudaEvent_t> d2hStop(streamCount);
563
564 std::vector<int> rowCounts(streamCount);
565 std::vector<int> rowOffsets(streamCount);
566
567 if (usePinned) {
568 CUDA_CHECK(cudaMallocHost(&pinned_B, matrixBytes));
569 std::copy(B.begin(), B.end(), pinned_B);
570 hostB = pinned_B;
571 }
572
573 // Создание streams, буферов и событий.
574 for (int i = 0; i < streamCount; ++i) {
575 const int rowOffset = i * rowsPerStream;
576 const int rows = std::max(0, std::min(rowsPerStream, N - rowOffset));
577
578 rowOffsets[i] = rowOffset;
579 rowCounts[i] = rows;
580
581 CUDA_CHECK(cudaStreamCreateWithFlags(&streams[i], cudaStreamNonBlocking));
582
583 CUDA_CHECK(cudaEventCreate(&h2dStart[i]));
584 CUDA_CHECK(cudaEventCreate(&h2dStop[i]));
585 CUDA_CHECK(cudaEventCreate(&kernelStart[i]));
586 CUDA_CHECK(cudaEventCreate(&kernelStop[i]));
587 CUDA_CHECK(cudaEventCreate(&d2hStart[i]));
588 CUDA_CHECK(cudaEventCreate(&d2hStop[i]));
589
590 if (rows == 0) {
591 continue;
592 }
593
594 const std::size_t elems = static_cast<std::size_t>(rows) * N;
595 if (usePinned) {
596 CUDA_CHECK(cudaMallocHost(&pinned_A[i], elems * sizeof(BASE_TYPE)));
597 CUDA_CHECK(cudaMallocHost(&pinned_C[i], elems * sizeof(BASE_TYPE)));
598 const BASE_TYPE* src = A.data() + static_cast<std::size_t>(rowOffset) * N;
599 std::copy(src, src + elems, pinned_A[i]);
600 }
601 if (algorithm == Algorithm::cuBLAS) {
602 CUBLAS_CHECK(cublasSetStream(handles[i], streams[i]));
603 }
604 }
605 CUDA_CHECK(cudaDeviceSynchronize());
606
607 CUDA_CHECK(cudaMalloc(&d_B, matrixBytes));
608 for (int i = 0; i < streamCount; ++i) {
609 if (rowCounts[i] == 0) continue;
610 const std::size_t chunkBytes =
611 static_cast<std::size_t>(rowCounts[i]) * N * sizeof(BASE_TYPE);
612 CUDA_CHECK(cudaMalloc(&d_A[i], chunkBytes));
613 CUDA_CHECK(cudaMalloc(&d_C[i], chunkBytes));
614 }
615 CUDA_CHECK(cudaDeviceSynchronize());
616
617 const auto bUploadStart = Clock::now();
618
619 CUDA_CHECK(cudaMemcpyAsync(
620 d_B,
621 hostB,
622 matrixBytes,
623 cudaMemcpyHostToDevice,
624 streams[0]
625 ));
626 CUDA_CHECK(cudaStreamSynchronize(streams[0]));
627
628 const double bUploadTime = elapsedMilliseconds(
629 bUploadStart, Clock::now()
630 );
631
632 // Запуск независимых частей вычисления.
633 for (int i = 0; i < streamCount; ++i) {
634 const int rows = rowCounts[i];
635 if (rows == 0) {
636 continue;
637 }
638
639 const int rowOffset = rowOffsets[i];
640
641 const BASE_TYPE* hostA = usePinned
642 ? pinned_A[i]
643 : A.data() + static_cast<std::size_t>(rowOffset) * N;
644
645 BASE_TYPE* hostC = usePinned
646 ? pinned_C[i]
647 : C.data() + static_cast<std::size_t>(rowOffset) * N;
648
649 const std::size_t chunkBytes =
650 static_cast<std::size_t>(rows) * N * sizeof(BASE_TYPE);
651
652 CUDA_CHECK(cudaEventRecord(h2dStart[i], streams[i]));
653
654 CUDA_CHECK(cudaMemcpyAsync(
655 d_A[i],
656 hostA,
657 chunkBytes,
658 cudaMemcpyHostToDevice,
659 streams[i]
660 ));
661
662 CUDA_CHECK(cudaEventRecord(h2dStop[i], streams[i]));
663 CUDA_CHECK(cudaEventRecord(kernelStart[i], streams[i]));
664
665 launchAlgorithm(
666 algorithm,
667 d_A[i],
668 d_B,
669 d_C[i],
670 N,
671 rows,
672 streams[i],
673 handles[i]
674 );
675
676 CUDA_CHECK(cudaEventRecord(kernelStop[i], streams[i]));
677 CUDA_CHECK(cudaEventRecord(d2hStart[i], streams[i]));
678
679 CUDA_CHECK(cudaMemcpyAsync(
680 hostC,
681 d_C[i],
682 chunkBytes,
683 cudaMemcpyDeviceToHost,
684 streams[i]
685 ));
686
687 CUDA_CHECK(cudaEventRecord(d2hStop[i], streams[i]));
688 }
689
690 // Ожидание завершения всех streams.
691 for (int i = 0; i < streamCount; ++i) {
692 CUDA_CHECK(cudaStreamSynchronize(streams[i]));
693 }
694
695 double maxH2D = 0.0;
696 double maxKernel = 0.0;
697 double maxD2H = 0.0;
698
699 for (int i = 0; i < streamCount; ++i) {
700 if (rowCounts[i] == 0) {
701 continue;
702 }
703
704 maxH2D = std::max(
705 maxH2D, eventElapsed(h2dStart[i], h2dStop[i])
706 );
707 maxKernel = std::max(
708 maxKernel, eventElapsed(kernelStart[i], kernelStop[i])
709 );
710 maxD2H = std::max(
711 maxD2H, eventElapsed(d2hStart[i], d2hStop[i])
712 );
713 }
714
715 timing.h2d = bUploadTime + maxH2D;
716 timing.kernel = maxKernel;
717 timing.d2h = maxD2H;
718
719 if (usePinned) {
720 for (int i = 0; i < streamCount; ++i) {
721 if (rowCounts[i] == 0) {
722 continue;
723 }
724
725 const std::size_t elements =
726 static_cast<std::size_t>(rowCounts[i]) * N;
727
728 const std::size_t offset =
729 static_cast<std::size_t>(rowOffsets[i]) * N;
730
731 std::copy(
732 pinned_C[i],
733 pinned_C[i] + elements,
734 C.begin() + static_cast<std::ptrdiff_t>(offset)
735 );
736 }
737 }
738
739 // Освобождение ресурсов.
740 for (int i = 0; i < streamCount; ++i) {
741 if (handles[i] != nullptr) {
742 CUBLAS_CHECK(cublasSetStream(handles[i], nullptr));
743 }
744
745 if (d_A[i] != nullptr) {
746 CUDA_CHECK(cudaFree(d_A[i]));
747 }
748 if (d_C[i] != nullptr) {
749 CUDA_CHECK(cudaFree(d_C[i]));
750 }
751
752 if (pinned_A[i] != nullptr) {
753 CUDA_CHECK(cudaFreeHost(pinned_A[i]));
754 }
755 if (pinned_C[i] != nullptr) {
756 CUDA_CHECK(cudaFreeHost(pinned_C[i]));
757 }
758
759 CUDA_CHECK(cudaEventDestroy(h2dStart[i]));
760 CUDA_CHECK(cudaEventDestroy(h2dStop[i]));
761 CUDA_CHECK(cudaEventDestroy(kernelStart[i]));
762 CUDA_CHECK(cudaEventDestroy(kernelStop[i]));
763 CUDA_CHECK(cudaEventDestroy(d2hStart[i]));
764 CUDA_CHECK(cudaEventDestroy(d2hStop[i]));
765
766 CUDA_CHECK(cudaStreamDestroy(streams[i]));
767 }
768
769 if (pinned_B != nullptr) {
770 CUDA_CHECK(cudaFreeHost(pinned_B));
771 }
772
773 CUDA_CHECK(cudaFree(d_B));
774
775 timing.total = elapsedMilliseconds(totalStart, Clock::now());
776 return timing;
777}

913Timing runOneCase(
914 Algorithm algorithm,
915 MemoryStrategy strategy,
916 const std::vector<BASE_TYPE>& A,
917 const std::vector<BASE_TYPE>& B,
918 std::vector<BASE_TYPE>& C,
919 int N)
920{

922 std::cerr << "Matrix size must be positive\n";
923 std::exit(EXIT_FAILURE);
924 }
925

927 static_cast<std::size_t>(N) * static_cast<std::size_t>(N);
928
929 if (A.size() != expectedSize ||
930 B.size() != expectedSize ||
931 C.size() != expectedSize) {
932 std::cerr << "Invalid matrix buffer size\n";
933 std::exit(EXIT_FAILURE);
934 }
935
936 switch (strategy) {
937 case MemoryStrategy::Pageable:
938 case MemoryStrategy::Pinned:
939 case MemoryStrategy::UnifiedPrefetch:
940 case MemoryStrategy::UnifiedManaged:
941 return runFullMatrix(
942 algorithm, strategy, A, B, C, N
943 );
944
945 case MemoryStrategy::PageableStreams:
946 case MemoryStrategy::PinnedStreams:
947 return runStreaming(
948 algorithm, strategy, A, B, C, N
949 );
950
951 default:
952 std::cerr << "Unknown memory strategy\n";
953 std::exit(EXIT_FAILURE);
954 }
955}

966double median(std::vector<double> values)
967{
968 if (values.empty()) {
969 return 0.0;
970 }
971
972 std::sort(values.begin(), values.end());
973
974 const std::size_t middle = values.size() / 2;
975
976 if (values.size() % 2 == 0) {
977 return (values[middle - 1] + values[middle]) / 2.0;
978 }
979
980 return values[middle];
981}
