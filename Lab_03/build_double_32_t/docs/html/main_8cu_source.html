52bool parsePositiveInt(std::string_view text, int& value) {
53 if (text.empty()) {
54 return false;
55 }
56
57 int parsedValue = 0;
58
59 const auto result = std::from_chars(
60 text.data(),
61 text.data() + text.size(),
62 parsedValue
63 );
64
65 if (result.ec != std::errc{} ||
66 result.ptr != text.data() + text.size() ||
67 parsedValue <= 0) {
68 return false;
69 }
70
71 value = parsedValue;
72 return true;
73}

109bool parseArguments(
110 int argc,
111 char** argv,
112 std::vector<int>& matrixSizes,
113 bool& singleRun,
114 int& blockSize
115) {
116 singleRun = false;
117 blockSize = TILE_SIZE;
118 matrixSizes.assign(
119 std::begin(MATRIX_SIZES),
120 std::end(MATRIX_SIZES)
121 );
122
123 if (argc == 1) {
124 return true;
125 }
126
127 const std::string_view first = argv[1];
128
129 if (argc == 2 && (first == "--help" || first == "-h")) {
130 printUsage(argv[0]);
131 return false;
132 }
133
134 if (argc > 3) {
135 std::cerr << "Error: invalid number of arguments.\n";
136 printUsage(argv[0]);
137 return false;
138 }
139
140 if (first != "all") {
141 int N = 0;
142
143 if (!parsePositiveInt(first, N)) {
144 std::cerr << "Error: N must be a positive integer or 'all'.\n";
145 printUsage(argv[0]);
146 return false;
147 }
148
149 if (N > MAX_SINGLE_MATRIX_SIZE) {
150 std::cerr << "Error: N must not exceed "
151 << MAX_SINGLE_MATRIX_SIZE << ".\n";
152 return false;
153 }
154
155 singleRun = true;
156 matrixSizes.assign(1, N);
157 }
158
159 if (argc == 3) {
160 int requested = 0;
161
162 if (!parsePositiveInt(argv[2], requested) ||
163 !isSupportedBlockSize(requested)) {
164 std::cerr << "Error: BLOCK_SIZE must be one of 2, 4, 8, 16, 32.\n";
165 return false;
166 }
167
168 blockSize = requested;
169 }
170
171 return true;
172}

199int main(int argc, char** argv) {
200#ifdef _WIN32
201 SetConsoleOutputCP(CP_UTF8);
202#endif
203
204 std::vector<int> matrixSizes;
205 bool singleRun = false;
206 int blockSize = TILE_SIZE;
207
208 if (!parseArguments(argc, argv, matrixSizes, singleRun, blockSize)) {
209 // --help также завершает работу через этот путь.
210 return (argc == 2 &&
211 (std::string_view(argv[1]) == "--help" ||
212 std::string_view(argv[1]) == "-h"))
213 ? EXIT_SUCCESS
214 : EXIT_FAILURE;
215 }
216 setBlockSize(blockSize);
217
218 std::srand(42);
219
220 std::cout << std::fixed << std::setprecision(6);
221
222 if (singleRun) {
223 std::cout << "\n[SINGLE-RUN MODE]\n";
224 } else {
225 std::cout << "\n[FULL BENCHMARK MODE]\n";
226 }
227
228 printGPUInfo();
229 std::cout << "Block size: " << blockSize << " x " << blockSize << '\n';
230 std::cout << "Using block size: " << getBlockSize() << '\n';
231
232 // Алгоритмы умножения матриц, участвующие в сравнении.
233 const std::vector<Algorithm> algorithms = {
234 Algorithm::Naive,
235 Algorithm::StaticTiled,
236 Algorithm::DynamicTiled,
237 Algorithm::cuBLAS
238 };
239
240 // Базовые стратегии размещения памяти и потоковой передачи данных.
241 std::vector<MemoryStrategy> strategies = {
242 MemoryStrategy::Pageable,
243 MemoryStrategy::Pinned,
244 MemoryStrategy::PageableStreams,
245 MemoryStrategy::PinnedStreams
246 };
247
248 // Unified Memory включается с учётом платформы и возможностей GPU.
249#ifdef __linux__
250 if (isUnifiedPrefetchSupported()) {
251 strategies.push_back(MemoryStrategy::UnifiedPrefetch);
252 } else {
253 std::cout
254 << "Skipping Unified Prefetch: "
255 << "GPU does not support concurrent managed access.\n";
256 }
257 strategies.push_back(MemoryStrategy::UnifiedManaged);
258#elif defined(_WIN32)
259 strategies.push_back(MemoryStrategy::UnifiedManaged);
260
261 std::cout
262 << "Note: Windows Unified Memory has limited support "
263 << "(concurrentManagedAccess = 0). "
264 << "Performance may be significantly lower.\n";
265#endif
266
267 // Раздельные файлы для полного и одиночного режимов.
268 const std::string csvFileName =
269 std::string(singleRun ? "cuda_lab3_single" : "cuda_lab3") +
270 (std::is_same_v<BASE_TYPE, float> ? "_f32" : "_f64") +
271 "_b" + std::to_string(getBlockSize()) +
272 (singleRun ? "_N" + std::to_string(matrixSizes.front()) : "") +
273 ".csv";
274
275 std::ofstream csv(csvFileName);
276
277 if (!csv.is_open()) {
278 std::cerr << "Cannot open CSV file: "
279 << csvFileName << '\n';
280 return EXIT_FAILURE;
281 }
282
283 writeCSVHeader(csv);
284
285 for (int N : matrixSizes) {
286 std::cout << "\n====================================\n";
287 std::cout << "MATRIX SIZE: " << N << " x " << N << '\n';
288 std::cout << "====================================\n";
289
290 const std::size_t matrixElements =
291 static_cast<std::size_t>(N) * static_cast<std::size_t>(N);
292
293 std::vector<BASE_TYPE> A(matrixElements);
294 std::vector<BASE_TYPE> B(matrixElements);
295 std::vector<BASE_TYPE> C(matrixElements);
296 std::vector<BASE_TYPE> C_reference;
297
298 initMatrix(A, N);
299 initMatrix(B, N);
300
301 // CPU-эталон вычисляется только для небольших матриц.
302 const bool useCPUReference = (N <= 1024);
303
304 if (useCPUReference) {
305 std::cout << "Computing CPU reference...\n";
306 computeCPU(A, B, C_reference, N);
307 } else {
308 std::cout
309 << "CPU full reference skipped (N > 1024).\n"
310 << "Independent sampled CPU verification will be used.\n";
311 }
312
313 for (MemoryStrategy strategy : strategies) {
314 std::cout << "\nMemory strategy: "
315 << strategyName(strategy)
316 << '\n';
317
318 for (Algorithm algorithm : algorithms) {
319 std::cout << " Algorithm: "
320 << algorithmName(algorithm)
321 << " ... "
322 << std::flush;
323
324 std::vector<double> h2dTimes;
325 std::vector<double> kernelTimes;
326 std::vector<double> d2hTimes;
327 std::vector<double> totalTimes;
328
329 bool verified = true;
330
331 // Прогревочные запуски не включаются в статистику.
332 for (int repeat = 0;
333 repeat < NUM_WARMUP + NUM_REPEATS;
334 ++repeat) {
335
336 std::fill(
337 C.begin(),
338 C.end(),
339 BASE_TYPE{0}
340 );
341
342 const Timing current = runOneCase(
343 algorithm,
344 strategy,
345 A,
346 B,
347 C,
348 N
349 );
350
351 if (repeat < NUM_WARMUP) {
352 continue;
353 }
354
355 h2dTimes.push_back(current.h2d);
356 kernelTimes.push_back(current.kernel);
357 d2hTimes.push_back(current.d2h);
358 totalTimes.push_back(current.total);
359 }
360
361 // Проверка результата после последнего измерительного запуска.
362 if (useCPUReference) {
363 verified = verifyResult(
364 C,
365 C_reference,
366 N
367 );
368 } else {
369 verified = verifySampled(
370 C,
371 A,
372 B,
373 N
374 );
375 }
376
377 BenchmarkResult result;
378
379 result.N = N;
380 result.blockSize = getBlockSize();
381 result.strategy = strategyName(strategy);
382 result.algorithm = algorithmName(algorithm);
383 result.precision = kPrecisionName;
384
385 result.h2dMedian = median(h2dTimes);
386 result.kernelMedian = median(kernelTimes);
387 result.d2hMedian = median(d2hTimes);
388 result.totalMedian = median(totalTimes);
389
390 result.totalMin = minimum(totalTimes);
391 result.verified = verified;
392
393 const double flops = 2.0 * N * N * N;
394 result.gflopsKernel = result.kernelMedian > 0.0
395 ? flops / (result.kernelMedian * 1.0e6) : 0.0;
396 result.gflopsTotal = result.totalMedian > 0.0
397 ? flops / (result.totalMedian * 1.0e6) : 0.0;
398
399 writeCSVRow(csv, result);
400 csv.flush();
401
402 std::cout
403 << (verified ? "OK" : "FAILED")
404 << " | H2D: " << result.h2dMedian
405 << " ms | Kernel: " << result.kernelMedian
406 << " ms (" << result.gflopsKernel << " GFLOPS)" << " | D2H: " << result.d2hMedian
407 << " ms | Total: " << result.totalMedian
408 << " ms\n";
409 }
410 }
411 }
412
413 csv.close();
414
415 std::cout << "\n====================================\n";
416 std::cout << "BENCHMARK FINISHED\n";
417 std::cout << "Results saved to " << csvFileName << '\n';
418 std::cout << "====================================\n";
419
420 return EXIT_SUCCESS;
421}
