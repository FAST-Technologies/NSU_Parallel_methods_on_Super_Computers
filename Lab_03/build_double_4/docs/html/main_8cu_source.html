46bool parsePositiveInt(std::string_view text, int& value) {
47 if (text.empty()) {
48 return false;
49 }
50
51 int parsedValue = 0;
52
53 const auto result = std::from_chars(
54 text.data(),
55 text.data() + text.size(),
56 parsedValue
57 );
58
59 if (result.ec != std::errc{} ||
60 result.ptr != text.data() + text.size() ||
61 parsedValue <= 0) {
62 return false;
63 }
64
65 value = parsedValue;
66 return true;
67}

103bool parseArguments(
104 int argc,
105 char** argv,
106 std::vector<int>& matrixSizes,
107 bool& singleRun,
108 int& blockSize
109) {
110 singleRun = false;
111 blockSize = TILE_SIZE;
112 matrixSizes.assign(
113 std::begin(MATRIX_SIZES),
114 std::end(MATRIX_SIZES)
115 );
116
117 if (argc == 1) {
118 return true;
119 }
120
121 const std::string_view first = argv[1];
122
123 if (argc == 2 && (first == "--help" || first == "-h")) {
124 printUsage(argv[0]);
125 return false;
126 }
127
128 if (argc > 3) {
129 std::cerr << "Error: invalid number of arguments.\n";
130 printUsage(argv[0]);
131 return false;
132 }
133
134 if (first != "all") {
135 int N = 0;
136
137 if (!parsePositiveInt(first, N)) {
138 std::cerr << "Error: N must be a positive integer or 'all'.\n";
139 printUsage(argv[0]);
140 return false;
141 }
142
143 if (N > MAX_SINGLE_MATRIX_SIZE) {
144 std::cerr << "Error: N must not exceed "
145 << MAX_SINGLE_MATRIX_SIZE << ".\n";
146 return false;
147 }
148
149 singleRun = true;
150 matrixSizes.assign(1, N);
151 }
152
153 if (argc == 3) {
154 int requested = 0;
155
156 if (!parsePositiveInt(argv[2], requested) ||
157 !isSupportedBlockSize(requested)) {
158 std::cerr << "Error: BLOCK_SIZE must be one of 2, 4, 8, 16, 32.\n";
159 return false;
160 }
161
162 blockSize = requested;
163 }
164
165 return true;
166}

197
198 std::vector<int> matrixSizes;
199 bool singleRun = false;

202 if (!parseArguments(argc, argv, matrixSizes, singleRun, blockSize)) {
203 // --help также завершает работу через этот путь.
204 return (argc == 2 &&

211
212 std::srand(42);
213
214 std::cout << std::fixed << std::setprecision(6);
215
216 if (singleRun) {
217 std::cout << "\n[SINGLE-RUN MODE]\n";
218 } else {

221

224 std::cout << "Using block size: " << getBlockSize() << '\n';
225
226 // Алгоритмы умножения матриц, участвующие в сравнении.
227 const std::vector<Algorithm> algorithms = {
228 Algorithm::Naive,
229 Algorithm::StaticTiled,
230 Algorithm::DynamicTiled,
231 Algorithm::cuBLAS

236 MemoryStrategy::Pageable,
237 MemoryStrategy::Pinned,
238 MemoryStrategy::PageableStreams,
239 MemoryStrategy::PinnedStreams
240 };
241
242 // Unified Memory включается с учётом платформы и возможностей GPU.
243#ifdef __linux__
244 if (isUnifiedPrefetchSupported()) {
245 strategies.push_back(MemoryStrategy::UnifiedPrefetch);
246 } else {
247 std::cout
248 << "Skipping Unified Prefetch: "
249 << "GPU does not support concurrent managed access.\n";
250 }
251 strategies.push_back(MemoryStrategy::UnifiedManaged);
252#elif defined(_WIN32)
253 strategies.push_back(MemoryStrategy::UnifiedManaged);
254
255 std::cout
256 << "Note: Windows Unified Memory has limited support "
257 << "(concurrentManagedAccess = 0). "
258 << "Performance may be significantly lower.\n";
259#endif
260
261 // Раздельные файлы для полного и одиночного режимов.
262 const std::string csvFileName =
263 std::string(singleRun ? "cuda_lab3_single" : "cuda_lab3") +
264 (std::is_same_v<BASE_TYPE, float> ? "_f32" : "_f64") +
265 "_b" + std::to_string(getBlockSize()) +
266 (singleRun ? "_N" + std::to_string(matrixSizes.front()) : "") +
267 ".csv";
268
269 std::ofstream csv(csvFileName);
270
271 if (!csv.is_open()) {
272 std::cerr << "Cannot open CSV file: "
273 << csvFileName << '\n';
274 return EXIT_FAILURE;
275 }
276
277 writeCSVHeader(csv);
278
279 for (int N : matrixSizes) {
280 std::cout << "\n====================================\n";
281 std::cout << "MATRIX SIZE: " << N << " x " << N << '\n';
282 std::cout << "====================================\n";
283
284 const std::size_t matrixElements =
285 static_cast<std::size_t>(N) * static_cast<std::size_t>(N);
286
287 std::vector<BASE_TYPE> A(matrixElements);
288 std::vector<BASE_TYPE> B(matrixElements);
289 std::vector<BASE_TYPE> C(matrixElements);
290 std::vector<BASE_TYPE> C_reference;
291
292 initMatrix(A, N);
293 initMatrix(B, N);
294
295 // CPU-эталон вычисляется только для небольших матриц.
296 const bool useCPUReference = (N <= 1024);
297
298 if (useCPUReference) {
299 std::cout << "Computing CPU reference...\n";
300 computeCPU(A, B, C_reference, N);
301 } else {
302 std::cout
303 << "CPU full reference skipped (N > 1024).\n"
304 << "Independent sampled CPU verification will be used.\n";
305 }
306
307 for (MemoryStrategy strategy : strategies) {
308 std::cout << "\nMemory strategy: "
309 << strategyName(strategy)
310 << '\n';
311
312 for (Algorithm algorithm : algorithms) {
313 std::cout << " Algorithm: "
314 << algorithmName(algorithm)
315 << " ... "
316 << std::flush;
317
318 std::vector<double> h2dTimes;
319 std::vector<double> kernelTimes;
320 std::vector<double> d2hTimes;
321 std::vector<double> totalTimes;
322
323 bool verified = true;
324
325 // Прогревочные запуски не включаются в статистику.
326 for (int repeat = 0;
327 repeat < NUM_WARMUP + NUM_REPEATS;
328 ++repeat) {
329
330 std::fill(
331 C.begin(),
332 C.end(),
333 BASE_TYPE{0}
334 );
335
336 const Timing current = runOneCase(
337 algorithm,
338 strategy,
339 A,
340 B,
341 C,
342 N
343 );
344
345 if (repeat < NUM_WARMUP) {
346 continue;
347 }
348
349 h2dTimes.push_back(current.h2d);
350 kernelTimes.push_back(current.kernel);
351 d2hTimes.push_back(current.d2h);
352 totalTimes.push_back(current.total);
353 }
354
355 // Проверка результата после последнего измерительного запуска.
356 if (useCPUReference) {
357 verified = verifyResult(
358 C,
359 C_reference,
360 N
361 );
362 } else {
363 verified = verifySampled(
364 C,
365 A,
366 B,
367 N
368 );
369 }
370
371 BenchmarkResult result;
372
373 result.N = N;
374 result.blockSize = getBlockSize();
375 result.strategy = strategyName(strategy);
376 result.algorithm = algorithmName(algorithm);
377
378 result.h2dMedian = median(h2dTimes);
379 result.kernelMedian = median(kernelTimes);
380 result.d2hMedian = median(d2hTimes);
381 result.totalMedian = median(totalTimes);
382
383 result.totalMin = minimum(totalTimes);
384 result.verified = verified;
385
386 writeCSVRow(csv, result);
387 csv.flush();
388
389 std::cout
390 << (verified ? "OK" : "FAILED")
391 << " | H2D: " << result.h2dMedian
392 << " ms | Kernel: " << result.kernelMedian
393 << " ms | D2H: " << result.d2hMedian
394 << " ms | Total: " << result.totalMedian
395 << " ms\n";
396 }
397 }
398 }
399
400 csv.close();
401
402 std::cout << "\n====================================\n";
403 std::cout << "BENCHMARK FINISHED\n";
404 std::cout << "Results saved to " << csvFileName << '\n';
405 std::cout << "====================================\n";
406
407 return EXIT_SUCCESS;
408}
