13/*
14 We distinguish between record types and event types (kinds), and have a
15 small number of pre-defined record types, including a raw one. We keep all
16 records double-aligned for the sake of the double timestamp field. Lengths
17 are given in doubles. Log records will usually consist of a
18 CLOG_Rec_header_t followed by one of the types that follow it below,
19 but record types CLOG_Rec_endblock_t and CLOG_Rec_endlog_t consist of
20 the header alone.
21*/

23/*
24 Whenever fields are add/deleted or padding is added or made meaningful,
25 synchronize the changes with all CLOG_Rec_xxx_swap_bytes() routines.
26*/
27typedef struct {
28 CLOG_Time_t time; /* Crucial to be the 1st item */
29 CLOG_CommLID_t icomm; /* LOCAL intracomm's internal local/global ID */
30 CLOG_int32_t rank; /* rank within communicator labelled by icomm */
31 CLOG_int32_t thread; /* local thread ID */
32 CLOG_int32_t rectype;
