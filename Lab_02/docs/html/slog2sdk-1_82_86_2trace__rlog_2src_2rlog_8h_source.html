235#define RLOG_LogEvent(pRLOG, event_param, starttime, endtime, recursion_param) \
236if (pRLOG->bLogging) \
237{ \
238 if (pRLOG->pOutput->pCurHeader + sizeof(RLOG_HEADER) + sizeof(RLOG_EVENT) > pRLOG->pOutput->pEnd) \
239 { \
240 WriteCurrentDataAndLogEvent(pRLOG, event_param, starttime, endtime, recursion_param); \
241 } \
242 else \
243 { \
244 RLOG_HEADER_CAST()->type = RLOG_EVENT_TYPE; \
245 RLOG_HEADER_CAST()->length = sizeof(RLOG_HEADER) + sizeof(RLOG_EVENT); \
246 RLOG_EVENT_CAST()->rank = pRLOG->nRank; \
247 RLOG_EVENT_CAST()->end_time = endtime - pRLOG->dFirstTimestamp; \
248 RLOG_EVENT_CAST()->start_time = starttime - pRLOG->dFirstTimestamp; \
249 RLOG_EVENT_CAST()->event = event_param; \
250 RLOG_EVENT_CAST()->recursion = recursion_param; \
251 /* advance the current position pointer */ \
252 pRLOG->pOutput->pCurHeader += sizeof(RLOG_HEADER) + sizeof(RLOG_EVENT); \
253 } \
254}
