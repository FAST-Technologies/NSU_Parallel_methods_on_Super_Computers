51 This assumes that sizeof(short) % sizeof(int) == 0.

136#define MPE_Log_TRAVERSE_LOG(condition) { \
137while (readBlk) { /* loop through the linked list of blocks */ \
138 n = readBlk->size; /* get # of ints in this block */ \
139 readRecHdr = (MPE_Log_HEADER *)(readBlk + 1); \
140 /* goto first record in this block */ \

146 i += readRecHdr->len; /* update used int count */ \

149 } \
